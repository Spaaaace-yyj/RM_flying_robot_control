cmake_minimum_required(VERSION 3.8)
project(ego_depth_inputs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(EGO_DEPTH_INPUTS_CORE_ONLY "Build geometry/timing tests without ROS 2" OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(depth_inputs_core_test test/test_core.cpp)
  target_include_directories(depth_inputs_core_test PRIVATE include)
  target_compile_options(depth_inputs_core_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME depth_inputs_core COMMAND depth_inputs_core_test)
endif()

if(NOT EGO_DEPTH_INPUTS_CORE_ONLY)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rcl_interfaces REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(message_filters REQUIRED)
  find_package(tf2 REQUIRED)
  find_package(tf2_ros REQUIRED)
  find_package(tf2_geometry_msgs REQUIRED)

  add_executable(ego_depth_inputs_node src/ego_depth_inputs_node.cpp)
  target_include_directories(ego_depth_inputs_node PRIVATE include)
  target_compile_options(ego_depth_inputs_node PRIVATE -Wall -Wextra -Wpedantic)
  ament_target_dependencies(ego_depth_inputs_node
    rclcpp rcl_interfaces sensor_msgs geometry_msgs nav_msgs std_msgs message_filters
    tf2 tf2_ros tf2_geometry_msgs)
  install(TARGETS ego_depth_inputs_node DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
  ament_package()
endif()
