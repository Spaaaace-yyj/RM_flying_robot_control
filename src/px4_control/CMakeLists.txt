cmake_minimum_required(VERSION 3.8)
project(px4_control LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PX4_CONTROL_CORE_ONLY "Build/test the C++ core without ROS2" OFF)
find_package(Eigen3 3.3 QUIET NO_MODULE)
if(NOT Eigen3_FOUND)
  find_path(EIGEN3_INCLUDE_DIR Eigen/Core PATH_SUFFIXES eigen3)
  if(NOT EIGEN3_INCLUDE_DIR)
    message(FATAL_ERROR "Install libeigen3-dev or provide EIGEN3_INCLUDE_DIR")
  endif()
  add_library(Eigen3::Eigen INTERFACE IMPORTED)
  set_target_properties(Eigen3::Eigen PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${EIGEN3_INCLUDE_DIR}")
endif()
add_library(px4_control_core STATIC src/core.cpp src/ego_trajectory.cpp)
target_include_directories(px4_control_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(px4_control_core PUBLIC Eigen3::Eigen)
target_compile_options(px4_control_core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(hover_core_test test/test_core.cpp)
  target_link_libraries(hover_core_test PRIVATE px4_control_core)
  target_compile_options(hover_core_test PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(ego_execution_test test/test_ego_execution.cpp)
  target_link_libraries(ego_execution_test PRIVATE px4_control_core)
  target_compile_options(ego_execution_test PRIVATE -Wall -Wextra -Wpedantic)
  if(PX4_CONTROL_CORE_ONLY)
    add_test(NAME hover_core COMMAND hover_core_test)
    add_test(NAME ego_execution COMMAND ego_execution_test)
    set(EGO_BSPLINE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../ego-planner-swarm/src/planner/bspline_opt")
    if(EXISTS "${EGO_BSPLINE_DIR}/include/bspline_opt/terminal_constraints.hpp")
      add_executable(ego_planner_contract_test test/test_ego_contract.cpp
        "${EGO_BSPLINE_DIR}/src/uniform_bspline.cpp")
      target_include_directories(ego_planner_contract_test PRIVATE "${EGO_BSPLINE_DIR}/include")
      target_link_libraries(ego_planner_contract_test PRIVATE px4_control_core)
      add_test(NAME ego_planner_contract COMMAND ego_planner_contract_test)
    endif()
  endif()
endif()
if(NOT PX4_CONTROL_CORE_ONLY)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rcl_interfaces REQUIRED)
  find_package(px4_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(std_srvs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(traj_utils REQUIRED)
  add_executable(px4_control_node src/px4_control_node.cpp)
  target_link_libraries(px4_control_node PRIVATE px4_control_core)
  ament_target_dependencies(px4_control_node PUBLIC rclcpp rcl_interfaces px4_msgs nav_msgs std_msgs std_srvs geometry_msgs traj_utils)
  target_compile_options(px4_control_node PRIVATE -Wall -Wextra -Wpedantic)
  if(BUILD_TESTING)
    find_package(ament_cmake_test REQUIRED)
    ament_add_test(hover_core_cpp COMMAND $<TARGET_FILE:hover_core_test>
      GENERATE_RESULT_FOR_RETURN_CODE_ZERO)
    ament_add_test(ego_execution_cpp COMMAND $<TARGET_FILE:ego_execution_test>
      GENERATE_RESULT_FOR_RETURN_CODE_ZERO)
  endif()
  install(TARGETS px4_control_node DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
  ament_package()
endif()
